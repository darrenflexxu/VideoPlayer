CMAKE_MINIMUM_REQUIRED(VERSION 3.16)

PROJECT(VideoProc)

set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 14)

set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${PROJECT_SOURCE_DIR}/Lib)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${PROJECT_SOURCE_DIR}/Lib)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${PROJECT_SOURCE_DIR}/Bin)

set(ThirdParty "${PROJECT_SOURCE_DIR}/thirdparty")

IF(WIN32)
SET(CMAKE_PREFIX_PATH C:/Qt/Qt6/6.x.x/msvc2019_64/lib/cmake;${ThirdParty}/opencv/lib)
ELSE(WIN32)
SET(CMAKE_PREFIX_PATH /usr/local/Qt6/lib/cmake;/opt/homebrew/opt/opencv/lib/cmake;/opt/homebrew/lib/cmake)
set(CMAKE_C_FLAGS "${CMAKE_C_FLAGS}  -fPIC")
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS}  -fPIC")
ENDIF()

find_package(OpenCV REQUIRED)

set(CMAKE_INCLUDE_CURRENT_DIR ON)
SET(CMAKE_AUTOMOC ON)
SET(CMAKE_AUTORCC ON)
SET(CMAKE_AUTOUIC ON)

FIND_PACKAGE(Qt6Widgets REQUIRED)
FIND_PACKAGE(Qt6Gui REQUIRED)
FIND_PACKAGE(Qt6Core REQUIRED)

IF(WIN32)
FIND_PACKAGE(Qt6OpenGL REQUIRED)
ENDIF()

FIND_PACKAGE(Qt6LinguistTools REQUIRED)
FIND_PACKAGE(Qt6Qml REQUIRED)
FIND_PACKAGE(Qt6Quick REQUIRED)

add_subdirectory(xcodec)
add_subdirectory(XVideoView)
add_subdirectory(XVideoTranscode)
add_subdirectory(xvideotranscode_cli)
